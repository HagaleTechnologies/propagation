**Goal:** Extend the pipeline to all 11 HF bands and horizons {0, 1h, 3h, 6h, 12h, 24h}, add RBN + live PSKReporter + optional cqdx R2 extractors, prove the public-only rerun still clears M2's bar, produce storm case studies, and gate on the ARCHITECTURE §6 minimum credible result.

Training scales to one `GBTModel` per horizon with band as a feature; storm case studies and the minimum-credible-result gate consume the same pinned eval interfaces.

```python
# src/propagation/data/common.py
"""Helpers shared by extractors: band mapping and raw-schema conformance."""

from __future__ import annotations

import polars as pl

from propagation.schema import SPOT_SCHEMA

# Amateur allocations (ITU Region-agnostic outer envelope), Hz.
BAND_EDGES_HZ: list[tuple[str, int, int]] = [
    ("160m", 1_800_000, 2_000_000),
    ("80m", 3_500_000, 4_000_000),
    ("60m", 5_250_000, 5_450_000),
    ("40m", 7_000_000, 7_300_000),
    ("30m", 10_100_000, 10_150_000),
    ("20m", 14_000_000, 14_350_000),
    ("17m", 18_068_000, 18_168_000),
    ("15m", 21_000_000, 21_450_000),
    ("12m", 24_890_000, 24_990_000),
    ("10m", 28_000_000, 29_700_000),
    ("6m", 50_000_000, 54_000_000),
]


def band_from_freq_hz(freq_hz: int) -> str | None:
    for band, lo, hi in BAND_EDGES_HZ:
        if lo <= freq_hz <= hi:
            return band
    return None


def conform_spots(df: pl.DataFrame) -> pl.DataFrame:
    """Cast/fill/reorder df to exactly SPOT_SCHEMA."""
    exprs = []
    for name, dtype in SPOT_SCHEMA.items():
        col = pl.col(name) if name in df.columns else pl.lit(None)
        exprs.append(col.cast(dtype).alias(name))
    return df.select(exprs)
```

```python
# src/propagation/data/rbn.py
"""RBN daily-archive extractor.

The RBN raw archive (one zip per UTC day) carries NO location for either
side of a spot. Spotter (de) grids are joined in from the RBN nodes list;
DX-side grids stay null and are filled by callsign-grid enrichment
(SPEC-labeling §1.6) before hygiene. WPM (`speed`) has no SPOT_SCHEMA
column and is dropped.
"""

from __future__ import annotations

import argparse
import io
import time
import zipfile
from datetime import date, timedelta
from pathlib import Path
from typing import Callable

import httpx
import polars as pl

from propagation.data.common import band_from_freq_hz, conform_spots
from propagation.data.lake import Lake

ARCHIVE_URL = "https://www.reversebeacon.net/raw_data/dl.php?f={ymd}"
NODES_URL = "https://www.reversebeacon.net/nodes/"  # verify format at execution
RBN_COLUMNS = [
    "callsign", "de_pfx", "de_cont", "freq", "band", "dx", "dx_pfx",
    "dx_cont", "mode", "db", "date", "speed", "tx_mode",
]
_PAUSE_S = 2.0


def _http_get(url: str) -> bytes:
    delay = 2.0
    for attempt in range(3):
        try:
            r = httpx.get(url, timeout=60, follow_redirects=True)
            r.raise_for_status()
            return r.content
        except (httpx.HTTPStatusError, httpx.TransportError):
            if attempt == 2:
                raise
            time.sleep(delay)
            delay *= 2
    raise AssertionError("unreachable")


def parse_rbn_csv(raw: bytes, spotter_grids: dict[str, str]) -> pl.DataFrame:
    df = pl.read_csv(io.BytesIO(raw), infer_schema_length=0)
    missing = [c for c in ("callsign", "freq", "dx", "mode", "db", "date") if c not in df.columns]
    if missing:
        raise ValueError(f"RBN CSV header changed; missing {missing}: got {df.columns}")
    df = df.filter(
        pl.col("freq").str.strip_chars().str.len_chars() > 0,
        pl.col("dx").str.strip_chars().str.len_chars() > 0,
    )
    freq_hz = (pl.col("freq").cast(pl.Float64) * 1000).round(0).cast(pl.Int64)
    out = df.select(
        pl.lit("rbn").alias("source"),
        pl.col("date").str.to_datetime("%Y-%m-%d %H:%M:%S", time_zone="UTC").alias("ts"),
        freq_hz.map_elements(band_from_freq_hz, return_dtype=pl.Utf8).alias("band"),
        pl.col("mode").alias("mode"),
        freq_hz.alias("freq_hz"),
        pl.col("dx").alias("dx_call"),
        pl.col("callsign").alias("de_call"),
        pl.lit(None, dtype=pl.Utf8).alias("dx_grid"),
        pl.col("callsign").replace_strict(spotter_grids, default=None).alias("de_grid"),
        pl.col("db").cast(pl.Int16, strict=False).alias("snr_db"),
    ).filter(pl.col("band").is_not_null())
    return conform_spots(out)


def load_spotter_grids(cache_dir: Path, http_get: Callable[[str], bytes] | None = None) -> dict[str, str]:
    """Fetch + cache the RBN nodes list; return callsign -> grid (as published).

    Execution-time verification required: confirm the nodes endpoint and its
    format (HTML table vs CSV), then adapt the parser here. Cached copy lives
    at cache_dir/rbn/nodes.csv as `callsign,grid` lines.
    """
    cache = cache_dir / "rbn" / "nodes.csv"
    if not cache.exists():
        get = http_get or _http_get
        raw = get(NODES_URL)
        cache.parent.mkdir(parents=True, exist_ok=True)
        cache.write_bytes(raw)
    grids: dict[str, str] = {}
    for line in cache.read_text().splitlines():
        parts = line.strip().split(",")
        if len(parts) >= 2 and parts[0] and parts[1]:
            grids[parts[0].upper()] = parts[1].upper()
    return grids


class RbnExtractor:
    source = "rbn"

    def __init__(self, spotter_grids: dict[str, str] | None = None):
        self._grids = spotter_grids

    def extract(
        self,
        start: date,
        end: date,
        bands: list[str],
        lake: Lake,
        cache_dir: Path,
        http_get: Callable[[str], bytes] | None = None,
    ) -> list[Path]:
        get = http_get or _http_get
        grids = self._grids if self._grids is not None else load_spotter_grids(cache_dir, get)
        written: list[Path] = []
        d = start
        while d <= end:
            ymd = d.strftime("%Y%m%d")
            cache = cache_dir / "rbn" / f"{ymd}.zip"
            if not cache.exists():
                cache.parent.mkdir(parents=True, exist_ok=True)
                cache.write_bytes(get(ARCHIVE_URL.format(ymd=ymd)))
                time.sleep(_PAUSE_S if http_get is None else 0)
            with zipfile.ZipFile(io.BytesIO(cache.read_bytes())) as z:
                csv_raw = z.read(z.namelist()[0])
            df = parse_rbn_csv(csv_raw, grids)
            for band in bands:
                part = df.filter(pl.col("band") == band, pl.col("ts").dt.date() == d)
                if part.height == 0:
                    continue
                written.append(lake.write_parquet(part, lake.spots_dir(band, d), name=f"rbn-{ymd}"))
            d += timedelta(days=1)
        return written


def main() -> None:
    ap = argparse.ArgumentParser(description="Extract RBN daily archives into the lake")
    ap.add_argument("--start", required=True, type=date.fromisoformat)
    ap.add_argument("--end", required=True, type=date.fromisoformat)
    ap.add_argument("--band", action="append", required=True, dest="bands")
    ap.add_argument("--lake-root", type=Path, default=Path("data/lake"))
    ap.add_argument("--cache-dir", type=Path, default=Path("data/cache"))
    args = ap.parse_args()
    files = RbnExtractor().extract(args.start, args.end, args.bands, Lake(args.lake_root), args.cache_dir)
    print(f"wrote {len(files)} files")


if __name__ == "__main__":
    main()
```

### Task 3: Callsign→grid enrichment (`data/callgrids.py` + SPEC v1.1 amendment)

RBN rows have no DX-side location, so without enrichment every RBN spot is
non-qualifying (SPEC §1.2) and RBN adds zero labels. Fix: a deterministic
per-UTC-date callsign→grid4 table built from that date's grid-bearing
qualifying spots, applied only where a side has null grid AND null lat/lon.
This changes label construction, so it MUST land as a normative SPEC
amendment first.

```markdown
### 1.6 Location enrichment (v1.1 addition)

Sources whose archives omit location (RBN) are enriched before rule 2:
for each UTC date, build a callsign→grid4 table from that date's qualifying
grid-bearing spots (union of (dx_call, dx_grid4) and (de_call, de_grid4)
pairs with non-null grid4, all sources, all bands); the table value is the
modal grid4 per callsign, ties broken by lexicographically smallest grid4.
A spot row with null grid AND null lat/lon on a side takes that side's
enriched grid4 (marked {dx,de}_grid_src='enriched'); rows still lacking
location remain non-qualifying. The enrichment table is deterministic, so
two implementations still produce identical label sets. Enriched-row
fraction is reported per band/day alongside the §6.8 counters.
```

```python
# src/propagation/data/callgrids.py
"""Deterministic callsign→grid4 enrichment (SPEC-labeling §1.6)."""

from __future__ import annotations

import polars as pl


def build_callgrids(spots_q: pl.DataFrame) -> pl.DataFrame:
    """Modal grid4 per callsign from grid-bearing qualified spots.

    Ties break to the lexicographically smallest grid4 (SPEC §1.6).
    """
    pairs = pl.concat(
        [
            spots_q.select(pl.col("dx_call").alias("call"), pl.col("dx_grid4").alias("grid4")),
            spots_q.select(pl.col("de_call").alias("call"), pl.col("de_grid4").alias("grid4")),
        ]
    ).drop_nulls()
    counts = pairs.group_by("call", "grid4").agg(pl.len().cast(pl.Int32).alias("n"))
    return (
        counts.sort(["call", "n", "grid4"], descending=[False, True, False])
        .group_by("call", maintain_order=True)
        .first()
    )


def _enrich_side(df: pl.DataFrame, side: str, table: pl.DataFrame) -> pl.DataFrame:
    call, grid, lat, lon = f"{side}_call", f"{side}_grid", f"{side}_lat", f"{side}_lon"
    lookup = dict(zip(table["call"], table["grid4"]))
    needs = pl.col(grid).is_null() & pl.col(lat).is_null() & pl.col(lon).is_null()
    filled = pl.col(call).replace_strict(lookup, default=None)
    return df.with_columns(
        pl.when(needs & filled.is_not_null())
        .then(pl.lit("enriched"))
        .when(pl.col(grid).is_not_null() | pl.col(lat).is_not_null())
        .then(pl.lit("reported"))
        .otherwise(pl.lit(None, dtype=pl.Utf8))
        .alias(f"{side}_grid_src"),
        pl.when(needs).then(filled).otherwise(pl.col(grid)).alias(grid),
    )


def enrich_grids(raw: pl.DataFrame, callgrids: pl.DataFrame) -> pl.DataFrame:
    """Fill null-location sides on raw-schema rows. Applied BEFORE hygiene."""
    return _enrich_side(_enrich_side(raw, "dx", callgrids), "de", callgrids)
```

```python
# src/propagation/data/pskreporter.py
"""Live PSKReporter MQTT accumulator.

Broker mqtt.pskreporter.info, topic tree
pskr/filter/v2/{band}/{mode}/{sendercall}/{receivercall}/{sendergrid}/
{receivergrid}/{senderDXCC}/{receiverDXCC}; JSON payloads use short keys
(sq f md rp t sc sl rc rl). Both facts flagged for execution-time
verification (see plan preamble); we subscribe to the whole v2 tree and
derive band from `f` client-side so topic-token semantics never matter.

Downtime handling: gaps produce no spots → receivers not provably
monitoring → windows unlabeled (SPEC-labeling §2/§3). Never fabricate
negatives for gap windows; nothing downstream needs to know about gaps.
"""

from __future__ import annotations

import argparse
import json
import signal
import sys
import time
from datetime import UTC, datetime
from pathlib import Path

import polars as pl

from propagation.data.common import band_from_freq_hz, conform_spots
from propagation.data.lake import Lake

DEFAULT_BROKER = "mqtt.pskreporter.info"
DEFAULT_TOPIC = "pskr/filter/v2/#"


def parse_payload(payload: bytes) -> dict | None:
    try:
        m = json.loads(payload)
        freq_hz = int(m["f"])
        band = band_from_freq_hz(freq_hz)
        if band is None:
            return None
        return {
            "source": "pskreporter",
            "ts": datetime.fromtimestamp(int(m["t"]), tz=UTC),
            "band": band,
            "mode": str(m["md"]),
            "freq_hz": freq_hz,
            "dx_call": str(m["sc"]),   # sender = transmitter
            "de_call": str(m["rc"]),   # receiver = spotter
            "dx_grid": m.get("sl"),
            "de_grid": m.get("rl"),
            "snr_db": int(m["rp"]) if m.get("rp") is not None else None,
        }
    except (ValueError, KeyError, TypeError, json.JSONDecodeError):
        return None


class PskAccumulator:
    def __init__(self, lake: Lake, flush_every_s: int = 3600):
        self.lake = lake
        self.flush_every_s = flush_every_s
        self._rows: list[dict] = []
        self._last_msg_ts: datetime | None = None

    @property
    def pending(self) -> int:
        return len(self._rows)

    def add(self, row: dict | None) -> None:
        if row is None:
            return
        if self._last_msg_ts is not None and (row["ts"] - self._last_msg_ts).total_seconds() > 900:
            print(f"gap: no messages {self._last_msg_ts.isoformat()} -> {row['ts'].isoformat()}", file=sys.stderr)
        self._last_msg_ts = row["ts"]
        self._rows.append(row)

    def flush(self, now: datetime) -> list[Path]:
        if not self._rows:
            return []
        df = conform_spots(pl.DataFrame(self._rows))
        self._rows = []
        stamp = now.strftime("%Y%m%dT%H%M%S")
        written: list[Path] = []
        for (band, day), part in df.group_by(["band", df["ts"].dt.date().alias("d")], maintain_order=True):
            written.append(self.lake.write_parquet(part.drop("d") if "d" in part.columns else part,
                                                   self.lake.spots_dir(band, day), name=f"pskr-{stamp}"))
        return written

    def run(self, broker: str = DEFAULT_BROKER, topic: str = DEFAULT_TOPIC) -> None:
        import paho.mqtt.client as mqtt

        client = mqtt.Client(mqtt.CallbackAPIVersion.VERSION2)
        client.on_message = lambda c, u, msg: self.add(parse_payload(msg.payload))
        client.on_connect = lambda c, u, flags, rc, props: c.subscribe(topic)
        stopping = False

        def stop(signum, frame):
            nonlocal stopping
            stopping = True

        signal.signal(signal.SIGTERM, stop)
        signal.signal(signal.SIGINT, stop)
        client.connect(broker, 1883, keepalive=60)
        client.loop_start()
        last_flush = time.monotonic()
        try:
            while not stopping:
                time.sleep(1)
                if time.monotonic() - last_flush >= self.flush_every_s:
                    self.flush(datetime.now(UTC))
                    last_flush = time.monotonic()
        finally:
            client.loop_stop()
            client.disconnect()
            self.flush(datetime.now(UTC))  # final flush on shutdown


def main() -> None:
    ap = argparse.ArgumentParser(description="Accumulate PSKReporter MQTT spots into the lake")
    ap.add_argument("--lake-root", type=Path, default=Path("data/lake"))
    ap.add_argument("--broker", default=DEFAULT_BROKER)
    ap.add_argument("--topic", default=DEFAULT_TOPIC)
    ap.add_argument("--flush-every-s", type=int, default=3600)
    args = ap.parse_args()
    PskAccumulator(Lake(args.lake_root), args.flush_every_s).run(args.broker, args.topic)


if __name__ == "__main__":
    main()
```

```python
# src/propagation/data/cqdx_r2.py
"""OPTIONAL private extractor: cqdx R2 spot archive (S3-compatible).

Accelerant only — published results must reproduce without it (the Task 7
public-only rerun enforces this). Env-gated; importing this module never
requires boto3 or credentials. Rows land with source='cqdx' regardless of
the record's inner `source` field; dedup priority puts cqdx last so public
rows win ties. Nothing downstream may branch on source beyond that column.
"""

from __future__ import annotations

import argparse
import gzip
import json
import os
import sys
from datetime import UTC, date, datetime, timedelta
from pathlib import Path

import polars as pl

from propagation.data.common import conform_spots
from propagation.data.lake import Lake

_ENV = ("CQDX_R2_ENDPOINT", "CQDX_R2_ACCESS_KEY", "CQDX_R2_SECRET", "CQDX_R2_BUCKET")


def _ts(v) -> datetime | None:
    """cqdx `timestamp`: ISO-8601 string or epoch (ms if > 1e12, else s)."""
    try:
        if isinstance(v, str):
            return datetime.fromisoformat(v.replace("Z", "+00:00")).astimezone(UTC)
        n = float(v)
        return datetime.fromtimestamp(n / 1000 if n > 1e12 else n, tz=UTC)
    except (ValueError, TypeError, OSError):
        return None


def parse_ndjson_gz(raw: bytes) -> pl.DataFrame:
    rows = []
    for line in gzip.decompress(raw).decode().splitlines():
        if not line.strip():
            continue
        try:
            m = json.loads(line)
        except json.JSONDecodeError:
            continue
        ts = _ts(m.get("timestamp"))
        if ts is None or not m.get("band"):
            continue
        rows.append(
            {
                "source": "cqdx",
                "ts": ts,
                "band": m["band"],
                "mode": m.get("mode"),
                "freq_hz": int(m["frequency"]) if m.get("frequency") is not None else None,
                "dx_call": m.get("dxCall"),
                "de_call": m.get("deCall"),
                "dx_grid": m.get("dxGrid"),
                "de_grid": m.get("deGrid"),
                "dx_lat": m.get("dxLat"),
                "dx_lon": m.get("dxLon"),
                "de_lat": m.get("deLat"),
                "de_lon": m.get("deLon"),
                "snr_db": int(m["snr"]) if m.get("snr") is not None else None,
            }
        )
    if not rows:
        return conform_spots(pl.DataFrame({"source": pl.Series([], dtype=pl.Utf8)}))
    return conform_spots(pl.DataFrame(rows))


def _client_from_env():
    missing = [k for k in _ENV if not os.environ.get(k)]
    if missing:
        print(f"extract-cqdx requires env vars: {' '.join(missing)}", file=sys.stderr)
        raise SystemExit(2)
    import boto3  # lazy: lives under the `cqdx` extra

    return (
        boto3.client(
            "s3",
            endpoint_url=os.environ["CQDX_R2_ENDPOINT"],
            aws_access_key_id=os.environ["CQDX_R2_ACCESS_KEY"],
            aws_secret_access_key=os.environ["CQDX_R2_SECRET"],
        ),
        os.environ["CQDX_R2_BUCKET"],
    )


class CqdxR2Extractor:
    source = "cqdx"

    def __init__(self, client=None, bucket: str | None = None):
        self._client = client
        self._bucket = bucket

    def extract(self, start: date, end: date, bands: list[str], lake: Lake, cache_dir: Path) -> list[Path]:
        client, bucket = (self._client, self._bucket) if self._client else _client_from_env()
        written: list[Path] = []
        d = start
        while d <= end:
            frames = []
            for hour in range(24):
                prefix = f"raw/{d.isoformat()}/{hour:02d}/"
                resp = client.list_objects_v2(Bucket=bucket, Prefix=prefix)
                for obj in resp.get("Contents", []):
                    body = client.get_object(Bucket=bucket, Key=obj["Key"])["Body"].read()
                    frames.append(parse_ndjson_gz(body))
            if frames:
                df = pl.concat([f for f in frames if f.height], how="vertical") if any(f.height for f in frames) else None
                if df is not None:
                    for band in bands:
                        part = df.filter(pl.col("band") == band, pl.col("ts").dt.date() == d)
                        if part.height:
                            written.append(
                                lake.write_parquet(part, lake.spots_dir(band, d), name=f"cqdx-{d:%Y%m%d}")
                            )
            d += timedelta(days=1)
        return written


def main() -> None:
    ap = argparse.ArgumentParser(description="Extract cqdx R2 archive (optional, private)")
    ap.add_argument("--start", required=True, type=date.fromisoformat)
    ap.add_argument("--end", required=True, type=date.fromisoformat)
    ap.add_argument("--band", action="append", required=True, dest="bands")
```

```python
# src/propagation/pipeline.py
"""Rebuild derived lake artifacts (spots_q, uptime, labels) from raw spots.

`sources=` restricts which extractors' rows participate — the open/closed
boundary as an executable check (--sources wsprnet,rbn,pskreporter rebuilds
a public-only lake). Uptime evidence intervals pad ±30 min across midnight,
so each day reads its neighbors' raw partitions too.
"""

from __future__ import annotations

import argparse
from datetime import date, datetime, time, timedelta, timezone
from pathlib import Path

import polars as pl

from propagation.data.callgrids import build_callgrids, enrich_grids
from propagation.data.hygiene import dedup_spots, qualify_spots
from propagation.data.lake import Lake
from propagation.features.labels import build_labels, build_receiver_uptime

HORIZONS_S: list[int] = [0, 3600, 10800, 21600, 43200, 86400]


def _read_raw(lake: Lake, band: str, days: list[date]) -> pl.DataFrame:
    frames = []
    for d in days:
        dir_ = lake.spots_dir(band, d)
        if dir_.exists():
            frames.append(pl.read_parquet(dir_ / "*.parquet"))
    if not frames:
        return pl.DataFrame()
    return pl.concat(frames, how="vertical")


def rebuild(
    raw_lake: Lake,
    out_lake: Lake,
    bands: list[str],
    start: date,
    end: date,
    sources: set[str] | None = None,
) -> None:
    d = start
    while d <= end:
        for band in bands:
            raw = _read_raw(raw_lake, band, [d - timedelta(days=1), d, d + timedelta(days=1)])
            if raw.height == 0:
                continue
            if sources is not None:
                raw = raw.filter(pl.col("source").is_in(sorted(sources)))
            if raw.height == 0:
                continue
            # SPEC §1.6: enrichment table from this date's grid-bearing qualified spots
            q_pre, _ = qualify_spots(raw)
            table = build_callgrids(q_pre)
            q, _rejects = qualify_spots(enrich_grids(raw, table))
            q = dedup_spots(q)
            day_start = datetime.combine(d, time(0), tzinfo=timezone.utc)
            day_end = day_start + timedelta(days=1)
            pad = timedelta(minutes=30)
            q_padded = q.filter(pl.col("ts") >= day_start - pad, pl.col("ts") < day_end + pad)
            q_day = q.filter(pl.col("ts") >= day_start, pl.col("ts") < day_end)
            if q_day.height == 0:
                continue
            out_lake.write_parquet(q_day, out_lake.spots_q_dir(band, d))
            out_lake.write_parquet(table, out_lake.callgrids_dir(d), name=f"band-{band}")
            uptime = build_receiver_uptime(q_padded, band, d)
            out_lake.write_parquet(uptime, out_lake.uptime_dir(band, d))
            labels = build_labels(q_day, uptime, band, d)
            out_lake.write_parquet(labels, out_lake.labels_dir(band, d))
        d += timedelta(days=1)


def main() -> None:
    from propagation.schema import BANDS

    ap = argparse.ArgumentParser(description="Rebuild spots_q/uptime/labels from raw spots")
    ap.add_argument("--start", required=True, type=date.fromisoformat)
    ap.add_argument("--end", required=True, type=date.fromisoformat)
    ap.add_argument("--band", action="append", dest="bands", default=None,
                    help="repeatable; default: all supported bands")
    ap.add_argument("--lake-root", type=Path, default=Path("data/lake"))
    ap.add_argument("--out-lake-root", type=Path, default=None,
                    help="write derived artifacts to a different lake (default: in place)")
    ap.add_argument("--sources", default=None,
                    help="comma-separated source filter, e.g. wsprnet,rbn,pskreporter")
    args = ap.parse_args()
    raw_lake = Lake(args.lake_root)
    out_lake = Lake(args.out_lake_root) if args.out_lake_root else raw_lake
    sources = set(args.sources.split(",")) if args.sources else None
    rebuild(raw_lake, out_lake, args.bands or BANDS, args.start, args.end, sources)
    print("rebuild complete")


if __name__ == "__main__":
    main()
```

```python
# src/propagation/train.py
"""Fit one GBTModel per horizon (band as a feature) and score all models
over eval folds, writing FULL-label-set predictions to
reports/predictions/{model_id}/band={band}/h={horizon_s}.parquet."""

from __future__ import annotations

import argparse
from datetime import date, datetime
from pathlib import Path

import polars as pl

from propagation.data.lake import Lake
from propagation.eval.splits import Fold, blocked_folds
from propagation.features.labels import sample_training
from propagation.features.matrix import feature_matrix
from propagation.models.climatology import ClimatologyModel
from propagation.models.gbt import GBTModel
from propagation.models.p533_baseline import P533Model
from propagation.pipeline import HORIZONS_S

CELL_COLS = ["window_start", "tx_field", "rx_field", "band"]


def _labels_between(lake: Lake, bands: list[str], start: datetime, end: datetime) -> pl.DataFrame:
    con = lake.connect()
    return con.execute(
        "SELECT * FROM labels WHERE band IN (SELECT unnest(?::VARCHAR[])) AND window_start >= ? AND window_start < ?",
        [bands, start, end],
    ).pl()


def _sampled(labels: pl.DataFrame) -> pl.DataFrame:
    parts = []
    for (band, d), grp in labels.group_by(
        [pl.col("band"), pl.col("window_start").dt.date().alias("d")], maintain_order=True
    ):
        parts.append(sample_training(grp.drop("d") if "d" in grp.columns else grp, band, d))
    return pl.concat(parts, how="vertical")


def train_all(
    lake: Lake,
    bands: list[str],
    folds: list[Fold],
    models_dir: Path,
    horizons_s: list[int] = HORIZONS_S,
    predictions_dir: Path = Path("reports/predictions"),
    ssn_by_month: dict[str, float] | None = None,
) -> None:
    fold0 = folds[0]
    train_labels = _labels_between(lake, bands, fold0.train_start, fold0.train_end)
    sampled = _sampled(train_labels)

    models: dict[str, dict[int, object]] = {}
    for h in horizons_s:
        feats = feature_matrix(lake, sampled, h)
        m = GBTModel(h).fit(sampled, feats)
        out = models_dir / m.model_id
        out.parent.mkdir(parents=True, exist_ok=True)
        m.save(out)
        models.setdefault("gbt", {})[h] = m

    clim = ClimatologyModel().fit(sampled)
    p533 = P533Model(ssn_by_month or {})
    baselines = {"climatology": clim, "p533": p533}

    for fi, fold in enumerate(folds):
        eval_labels = _labels_between(lake, bands, fold.eval_start, fold.eval_end)  # FULL set
        for band in bands:
            band_labels = eval_labels.filter(pl.col("band") == band)
            if band_labels.height == 0:
                continue
            cells = band_labels.select(CELL_COLS)
            truth = band_labels.select(CELL_COLS + ["open", "sample_weight"])
            for h in horizons_s:
                preds = models["gbt"][h].predict_p_open(cells)
                _write_preds(predictions_dir, models["gbt"][h].model_id, band, h, preds, truth, fi)
            for name, model in baselines.items():
                preds = model.predict_p_open(cells)
                # baselines are horizon-independent; record under h=0 only
                _write_preds(predictions_dir, name, band, 0, preds, truth, fi)


def _write_preds(root: Path, model_id: str, band: str, h: int,
                 preds: pl.DataFrame, truth: pl.DataFrame, fold: int) -> None:
    df = preds.join(truth, on=CELL_COLS, how="inner").with_columns(pl.lit(fold, dtype=pl.Int32).alias("fold"))
    dest = root / model_id / f"band={band}"
    dest.mkdir(parents=True, exist_ok=True)
    path = dest / f"h={h}.parquet"
    if path.exists():
        df = pl.concat([pl.read_parquet(path), df], how="vertical")
    df.write_parquet(path)


def main() -> None:
    from propagation.schema import BANDS

    ap = argparse.ArgumentParser(description="Train per-horizon GBTs and score all models over eval folds")
    ap.add_argument("--lake-root", type=Path, default=Path("data/lake"))
    ap.add_argument("--band", action="append", dest="bands", default=None)
    ap.add_argument("--data-start", required=True, type=datetime.fromisoformat)
    ap.add_argument("--data-end", required=True, type=datetime.fromisoformat)
    ap.add_argument("--eval-days", type=int, default=14)
    ap.add_argument("--models-dir", type=Path, default=Path("models/artifacts"))
    ap.add_argument("--predictions-dir", type=Path, default=Path("reports/predictions"))
    args = ap.parse_args()
    folds = blocked_folds(args.data_start, args.data_end, eval_days=args.eval_days)
    train_all(Lake(args.lake_root), args.bands or BANDS, folds, args.models_dir,
              predictions_dir=args.predictions_dir)
    print("training complete")


if __name__ == "__main__":
    main()
```

horizon-independent (climatology keys on hour/month; P.533 on month/SSN), so
their eval predictions are written once under `h=0` and reused for every

```python
# src/propagation/eval/storms.py
"""Storm-window case studies: select storms from definitive Kp and render
before/during/after field maps of model P(open) vs P.533 vs realized labels.

Definitive Kp is permitted here: storm selection is eval stratification,
never a model feature (SPEC-labeling §6.4).
"""

from __future__ import annotations

from dataclasses import dataclass
from datetime import datetime, timedelta
from pathlib import Path

import numpy as np
import polars as pl

CELL_COLS = ["window_start", "tx_field", "rx_field", "band"]


@dataclass(frozen=True)
class StormWindow:
    peak_ts: datetime
    peak_kp: float
    phases: dict[str, tuple[datetime, datetime]]


def _phases(peak: datetime) -> dict[str, tuple[datetime, datetime]]:
    h = timedelta(hours=1)
    return {
        "before": (peak - 48 * h, peak - 24 * h),
        "during": (peak - 12 * h, peak + 12 * h),
        "after": (peak + 24 * h, peak + 48 * h),
    }


def select_storm_windows(lake, min_kp: float = 7.0, n: int = 3, separation_days: int = 10) -> list[StormWindow]:
    con = lake.connect()
    kp = con.execute(
        "SELECT ts, kp_definitive FROM space_weather WHERE kp_definitive >= ? ORDER BY kp_definitive DESC",
        [min_kp],
    ).pl()
    picked: list[StormWindow] = []
    for row in kp.iter_rows(named=True):
        ts = row["ts"]
        if any(abs((ts - p.peak_ts).total_seconds()) < separation_days * 86400 for p in picked):
            continue
        picked.append(StormWindow(peak_ts=ts, peak_kp=row["kp_definitive"], phases=_phases(ts)))
        if len(picked) == n:
            break
    return picked


def field_map(df: pl.DataFrame, value_col: str) -> np.ndarray:
    """18x18 grid (rows=lat field A..R, cols=lon field A..R) of mean value_col by rx_field."""
    m = np.full((18, 18), np.nan)
    agg = df.group_by("rx_field").agg(pl.col(value_col).mean())
    for f, v in agg.iter_rows():
        col = ord(f[0]) - ord("A")   # first char = longitude field
        row = ord(f[1]) - ord("A")   # second char = latitude field
        if 0 <= row < 18 and 0 <= col < 18:
            m[row, col] = v
    return m


def render_case_study(lake, storm: StormWindow, band: str, tx_field: str,
                      models: dict[str, object], out_dir: Path) -> Path:
    import matplotlib

    matplotlib.use("Agg")
    import matplotlib.pyplot as plt

    con = lake.connect()
    dest = out_dir / f"storm-{storm.peak_ts:%Y%m%d}"
    dest.mkdir(parents=True, exist_ok=True)
    phase_names = ["before", "during", "after"]
    col_names = list(models.keys()) + ["observed"]
    fig, axes = plt.subplots(len(phase_names), len(col_names),
                             figsize=(4 * len(col_names), 3.2 * len(phase_names)), squeeze=False)
    stats: dict[str, dict[str, float]] = {}
    for i, phase in enumerate(phase_names):
        t0, t1 = storm.phases[phase]
        labels = con.execute(
            "SELECT * FROM labels WHERE band=? AND tx_field=? AND window_start >= ? AND window_start < ?",
            [band, tx_field, t0, t1],
        ).pl()
        cells = labels.select(CELL_COLS)
        panels = {name: m.predict_p_open(cells).rename({"p_open": "v"}) for name, m in models.items()}
        panels["observed"] = labels.select(CELL_COLS + [pl.col("open").cast(pl.Float64).alias("v")])
        stats[phase] = {}
        for j, name in enumerate(col_names):
            grid = field_map(panels[name], "v")
            ax = axes[i][j]
            im = ax.imshow(grid, origin="lower", vmin=0, vmax=1, cmap="viridis")
            ax.set_title(f"{name} — {phase}", fontsize=9)
            ax.set_xticks([]), ax.set_yticks([])
            stats[phase][name] = float(np.nanmean(grid)) if not np.all(np.isnan(grid)) else float("nan")
        fig.colorbar(im, ax=axes[i][-1], fraction=0.046)
    fig.suptitle(f"{band} from {tx_field} — Kp {storm.peak_kp} storm, peak {storm.peak_ts:%Y-%m-%d %H%MZ}")
    png = dest / "maps.png"
    fig.savefig(png, dpi=150, bbox_inches="tight")
    plt.close(fig)

    lines = [
        f"# Storm case study — peak Kp {storm.peak_kp} at {storm.peak_ts:%Y-%m-%d %H%MZ}",
        "",
        f"Band {band}, transmit field {tx_field}. Mean P(open)/open-rate across mapped rx fields:",
        "",
        "| phase | " + " | ".join(col_names) + " |",
        "|---" * (len(col_names) + 1) + "|",
    ]
    for phase in phase_names:
        lines.append("| " + phase + " | " + " | ".join(f"{stats[phase][n]:.3f}" for n in col_names) + " |")
    lines += [
        "",
        "Reading: P.533 (monthly-median climatology) cannot move between phases;",
        "the learned model should track the observed collapse during the storm",
        "and the recovery after. Divergence between the gbt and observed columns",
        "during the storm bounds what the model actually captured.",
    ]
    (dest / "narrative.md").write_text("\n".join(lines) + "\n")
    return png
```

```python
# src/propagation/eval/gate.py
"""M3 acceptance gate: the ARCHITECTURE §6 minimum credible result, exact."""

from __future__ import annotations

import argparse
from dataclasses import dataclass, field
from pathlib import Path

import polars as pl

from propagation.eval.metrics import brier

GATE_BANDS = ["20m", "15m", "10m"]
GATE_HORIZONS_S = [0, 10800]


@dataclass
class GateResult:
    passed: bool
    table: pl.DataFrame
    months: int
    storm_folds: int
    failures: list[str] = field(default_factory=list)


def _load(predictions_dir: Path, model_id: str, band: str, h: int) -> pl.DataFrame:
    path = predictions_dir / model_id / f"band={band}" / f"h={h}.parquet"
    if not path.exists():
        raise FileNotFoundError(f"missing predictions: {path}")
    return pl.read_parquet(path)


def _brier_of(df: pl.DataFrame) -> float:
    scored = df.filter(pl.col("p_open").is_not_null())
    return brier(scored["open"].to_numpy(), scored["p_open"].to_numpy())


def check_gate(predictions_dir: Path, lake) -> GateResult:
    failures: list[str] = []
    rows = []
    all_windows = []
    fold_spans: dict[int, tuple] = {}
    for band in GATE_BANDS:
        for h in GATE_HORIZONS_S:
            gbt = _load(predictions_dir, f"gbt-h{h}", band, h)
            p533 = _load(predictions_dir, "p533", band, 0)
            clim = _load(predictions_dir, "climatology", band, 0)
            b_gbt, b_p533, b_clim = _brier_of(gbt), _brier_of(p533), _brier_of(clim)
            rows.append({"band": band, "horizon_s": h, "gbt": b_gbt, "p533": b_p533, "climatology": b_clim})
            if not (b_gbt < b_p533 and b_gbt < b_clim):
                failures.append(
                    f"{band} h={h}: gbt Brier {b_gbt:.4f} not < p533 {b_p533:.4f} and climatology {b_clim:.4f}"
                )
            all_windows.append(gbt.select("window_start", "fold"))
    win = pl.concat(all_windows, how="vertical")
    months = win["window_start"].dt.strftime("%Y-%m").unique().len()
    if months < 3:
        failures.append(f"eval spans {months} months; need >= 3")
    con = lake.connect()
    storm_folds = 0
    for (fi,), grp in win.group_by(["fold"]):
        t0, t1 = grp["window_start"].min(), grp["window_start"].max()
        n = con.execute(
            "SELECT count(*) FROM space_weather WHERE kp_definitive >= 5 AND ts >= ? AND ts <= ?", [t0, t1]
        ).fetchone()[0]
        if n > 0:
            storm_folds += 1
    if storm_folds < 1:
        failures.append("no storm fold: no eval fold overlaps any 3-h definitive Kp >= 5")
    return GateResult(passed=not failures, table=pl.DataFrame(rows),
                      months=months, storm_folds=storm_folds, failures=failures)


def main() -> None:
    from propagation.data.lake import Lake

    ap = argparse.ArgumentParser(description="Check the M3 minimum-credible-result gate")
    ap.add_argument("--predictions-dir", type=Path, default=Path("reports/predictions"))
    ap.add_argument("--lake-root", type=Path, default=Path("data/lake"))
    ap.add_argument("--report", type=Path, default=Path("reports/m3-gate.md"))
    args = ap.parse_args()
    res = check_gate(args.predictions_dir, Lake(args.lake_root))
    verdict = "PASS" if res.passed else "FAIL"
    md = [f"# M3 gate: {verdict}", "",
          f"Eval months: {res.months} (need >=3); storm folds: {res.storm_folds} (need >=1)", "",
          res.table.to_pandas().to_markdown(index=False), ""]
    if res.failures:
        md += ["## Failures", ""] + [f"- {f}" for f in res.failures] + [""]
    args.report.parent.mkdir(parents=True, exist_ok=True)
    args.report.write_text("\n".join(md))
    print("\n".join(md))
    raise SystemExit(0 if res.passed else 1)


if __name__ == "__main__":
    main()
```
